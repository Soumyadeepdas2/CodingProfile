vector<int> gradingStudents(vector<int> grades) {
vector<int>ans;
for(int i = 0; i < grades.size();i++){
    if((grades[i])<38){
        ans.push_back(grades[i]);
    }else{
    int next = ((grades[i]/5)+1)*5;
    if(next-grades[i]<3){
        ans.push_back(next);
    }else if(next-grades[i]>=3){
        ans.push_back(grades[i]);
    } 
}}
return ans;
}
