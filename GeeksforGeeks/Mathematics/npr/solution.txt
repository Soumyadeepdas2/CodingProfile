class Solution {
  public:
    long long nPr(int n, int r) {
        // code here
        long long ans = 1;
        long long ans1 = 1;
        for(int i = n; i>1; i--){
            ans = ans*i;
        }
        for(int i = n-r; i>1; i--){
            ans1 = ans1*i;
        }
        return(ans/ans1);
    }
};
