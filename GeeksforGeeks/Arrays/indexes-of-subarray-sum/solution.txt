class Solution {
  public:
    vector<int> subarraySum(vector<int> &arr, int target) {
        // code here
        int left = 0;
        int right = 0;
        int sum = 0;
        for( right = 0; right < arr.size(); right++){
            sum += arr[right];
             while(sum>target){
                 sum -= arr[left];
                 left++;
             }
             if(sum==target){
                 return{ left+1, right+1};
             }
        }
        return{-1};
    }
};
