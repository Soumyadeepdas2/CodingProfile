class Solution {
  public:
    void insertAtIndex(vector<int> &arr, int index, int val) {
        // code here
        vector<int>ans;
        for(int i = 0; i < index;i++){
            ans.push_back(arr[i]);
        }
        ans.push_back(val);
       for(int i = index; i<arr.size(); i++){
           ans.push_back(arr[i]);
       }
      arr = ans;
      }
    
};
