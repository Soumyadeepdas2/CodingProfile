class Solution {
  public:
    vector<int> getMoreAndLess(vector<int> &arr, int target) {
        // code here
      
        
        int count = 0;
        int count1 =0;
        for(int i = 0; i < arr.size(); i++){
            if(arr[i]<=target){
                count++;
            }
            if(arr[i]>=target){
                count1++;
            }
        }
       vector<int> ans = {count, count1};
       return ans;
    }
};
