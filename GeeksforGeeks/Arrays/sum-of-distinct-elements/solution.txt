class Solution {
  public:
    int findSum(vector<int>& arr) {
        // code here
        sort(arr.begin(),arr.end());
        int sum = 0;
        int alt = 0;
        
        for(int i = 0;  i < arr.size();i++){
            if(alt!=arr[i]){
                alt = arr[i];
                sum += arr[i];
            }
        }
        return sum;
    }
};
